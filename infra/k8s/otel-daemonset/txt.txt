3) Add the transform processor:

transform/normalize_logs:
  error_mode: ignore
  log_statements:
    - statements:
        - merge_maps(log.cache, ParseJSON(log.body.string), "upsert")
          where IsMatch(log.body.string, "^\\s*\\{")

        - set(log.attributes["level"], ConvertCase(log.cache["level"], "lower"))
          where log.attributes["level"] == nil and IsString(log.cache["level"])

        - set(log.attributes["message"], log.cache["message"])
          where log.attributes["message"] == nil and IsString(log.cache["message"])

        - set(log.attributes["upstream"], log.cache["upstream"])
          where log.attributes["upstream"] == nil and IsString(log.cache["upstream"])

        - set(log.attributes["state"], log.cache["state"])
          where log.attributes["state"] == nil and IsString(log.cache["state"])

        - set(log.attributes["service"], log.cache["service"])
          where log.attributes["service"] == nil and IsString(log.cache["service"])

        - set(log.attributes["service"], resource.attributes["service.name"])
          where log.attributes["service"] == nil and IsString(resource.attributes["service.name"])

        - set(log.attributes["service"], resource.attributes["k8s.deployment.name"])
          where log.attributes["service"] == nil and IsString(resource.attributes["k8s.deployment.name"])

        - set(log.attributes["level"], ConvertCase(log.severity_text, "lower"))
          where log.attributes["level"] == nil and log.severity_text != ""

        - set(log.attributes["level"], "error")
          where log.attributes["level"] == nil and log.severity_number >= SEVERITY_NUMBER_ERROR

        - set(log.attributes["level"], "warn")
          where log.attributes["level"] == nil and log.severity_number >= SEVERITY_NUMBER_WARN

        - set(log.attributes["level"], "info")
          where log.attributes["level"] == nil and log.severity_number >= SEVERITY_NUMBER_INFO

        - set(log.attributes["message"], log.body.string)
          where log.attributes["message"] == nil and IsString(log.body)

4) Add the routing connector:

connectors:
  routing/logs:
    error_mode: ignore
    default_pipelines:
      - logs/unclassified
    table:
      - condition: IsString(resource.attributes["k8s.deployment.name"]) and IsMatch(resource.attributes["k8s.deployment.name"], "{{ .Values.logRouting.postgresDeploymentRegex }}")
        pipelines: [logs/postgres]
      - condition: IsString(resource.attributes["k8s.deployment.name"]) and IsMatch(resource.attributes["k8s.deployment.name"], "{{ .Values.logRouting.valkeyDeploymentRegex }}")
        pipelines: [logs/valkey]
      - condition: IsString(resource.attributes["k8s.deployment.name"]) and IsMatch(resource.attributes["k8s.deployment.name"], "{{ .Values.logRouting.appDeploymentRegex }}")
        pipelines: [logs/app]
